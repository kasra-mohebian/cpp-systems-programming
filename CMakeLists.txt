cmake_minimum_required(VERSION 3.20)
project(cpp_systems_programming VERSION 1.0.0 LANGUAGES CXX)
if(NOT UNIX)
    message(FATAL_ERROR "The RAII module requires POSIX (Linux or macOS).")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(CPPSTUDY_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
option(CPPSTUDY_THREAD_SANITIZER "Enable ThreadSanitizer (separate from ASan)" OFF)
if(CPPSTUDY_SANITIZERS AND CPPSTUDY_THREAD_SANITIZER)
    message(FATAL_ERROR "Use separate builds for ASan and TSan")
endif()
if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    message(FATAL_ERROR "This edition supports GCC and Clang")
endif()
find_package(Threads REQUIRED)
add_library(cppstudy
    src/AST.cpp src/BinaryHeap.cpp src/BitSet.cpp src/ChainingHashTable.cpp
    src/CommandLine.cpp src/Complex.cpp src/EvaluationContext.cpp src/ObjectRepresentation.cpp
    src/PrintVisitor.cpp src/Rational.cpp src/SimpleVM.cpp src/SortPointers.cpp
    src/StringMatching.cpp src/TempDirectory.cpp src/TempFile.cpp src/Varint.cpp)
target_include_directories(cppstudy PUBLIC include)
target_link_libraries(cppstudy PUBLIC Threads::Threads)
target_compile_options(cppstudy PRIVATE -Wall -Wextra -Wpedantic)
if(CPPSTUDY_SANITIZERS)
    target_compile_options(cppstudy PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(cppstudy PUBLIC -fsanitize=address,undefined)
elseif(CPPSTUDY_THREAD_SANITIZER)
    target_compile_options(cppstudy PUBLIC -fsanitize=thread -fno-omit-frame-pointer)
    target_link_options(cppstudy PUBLIC -fsanitize=thread)
endif()
add_executable(cppstudy_demo examples/demo.cpp)
target_link_libraries(cppstudy_demo PRIVATE cppstudy)
include(CTest)
if(BUILD_TESTING)
    add_executable(cppstudy_tests tests/tests.cpp)
    target_link_libraries(cppstudy_tests PRIVATE cppstudy)
    target_compile_options(cppstudy_tests PRIVATE -Wall -Wextra -Wpedantic)
    foreach(suite algorithms containers numbers expressions vm resources concurrency)
        add_test(NAME ${suite} COMMAND cppstudy_tests ${suite})
        set_tests_properties(${suite} PROPERTIES TIMEOUT 60)
    endforeach()
endif()
